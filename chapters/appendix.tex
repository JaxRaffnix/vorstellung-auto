\begin{frame}{Erklärung: Zuschüsse und Abschreibungen}

    \textbf{Zuschüsse}
    \begin{itemize}
        \item Zweckgebunde Förderungen für Anlagen/Projekte, werden nicht zurückgezahlt.
        \item Werden in der Passiva als \texttt{Sonderposten für Zuschüsse} verbucht.
        \item Aktuelle Investitionszuschüsse werden in der G\&V als \texttt{Zuweisungen für Zuschüsse} vom Ertrag abgezogen. Dadurch wird verhindert, dass sie den Jahresgewinn künstlich aufblähen.
    \end{itemize}
    
    \vspace{0.2cm}
    \textbf{Abschreibungen}
    \begin{itemize}
        \item Entsprechen der Wertminderung des Anlagevermögens.
        \item Die \texttt{Auflösung der Zuschüsse} bringt das Geld als Ertrag in die G\&V zurück. Das gleicht den Aufwand der \texttt{Abschreibungen} im selben Jahr exakt aus.
    \end{itemize}

\end{frame}
